[UVa 1345 Jamie's Contact Groups](https://blog.csdn.net/metaphysis/article/details/156425072)

## 题目描述

$\texttt{Jamie}$ 的手机里存有一个很长的联系人列表，她希望能将这些联系人分成若干组，以便更快地查找。每个联系人可以属于多个备选组（输入中给出），但最终必须被分配到恰好一个组中。 $\texttt{Jamie}$ 希望使得 **所有组中人数最多的那个人数尽可能小** 。请你帮她找出这个最小值。

### 输入格式

输入包含最多 $20$ 个测试用例。  
每个测试用例第一行是两个整数 $N$ 和 $M$ ， $N$ 表示联系人数（ $N \le 1000$ ）， $M$ 表示组数（ $M \le 500$ ）。  
接下来 $N$ 行，每行先是一个长度不超过 $15$ 的只含字母的名字，然后是该联系人可以属于的组的编号（ $0$ 到 $M-1$ ），名字不会重复。  
输入以一行 `0 0` 结束。

### 输出格式

对每个测试用例，输出一个整数，表示 **最大组的最小可能人数** 。

## 题目分析

这是一个典型的 **二分图的多重匹配** 问题：

- 左部节点： $N$ 个联系人。
- 右部节点： $M$ 个组。
- 每个联系人只能被分配到一个组，但一个组可以容纳多个联系人。
- 目标：在满足每个联系人只能选择一个允许的组的前提下，使得 **所有组中的人数最大值最小**。

如果我们直接枚举每个组最多能有多少人，然后检查这个限制是否可行，就可以通过 **二分答案** 来加速求解。

**为什么可以二分？**

假设我们猜测最大组的大小为 $limit$ ，那么我们可以构建这样一个网络流模型：

1. 源点 $s$ 向每个联系人连一条容量为 $1$ 的边（每人只能分配一次）。
2. 每个联系人向其允许的组连一条容量为 $1$ 的边（每人只能去一个组）。
3. 每个组向汇点 $t$ 连一条容量为 $limit$ 的边（该组最多容纳 $limit$ 人）。
4. 计算从 $s$ 到 $t$ 的最大流：
   - 如果最大流等于 $N$ ，说明所有人都能分配成功，且每组人数不超过 $limit$ ， $limit$ 可行。
   - 否则 $limit$ 不可行。

由于 $N \le 1000$ ， $M \le 500$ ，组数较少，而二分答案的次数为 $O(\log N)$ ，每次建图跑最大流的复杂度是可以接受的。

## 算法步骤

1. 读入数据，记录每个联系人可以去的组。
2. 二分可能的 $limit$ ，范围是 $[0, N]$ 。  
   - 若 $limit = 0$ ，显然不可能（除非 $N = 0$ ）。  
   - 若 $limit = N$ ，肯定可行（所有人放同一组）。  
3. 对每个 $mid$ ，建图跑 **$\texttt{Dinic}$ 最大流** 检查可行性。
4. 输出最小的可行 $limit$ 。

## 复杂度分析

- 二分次数： $O(\log N)$
- 每次 Dinic 的复杂度： $O(E \sqrt{V})$ 或 $O(EV^{2/3})$ 在单位容量图中表现良好，这里 $V = N + M + 2$ ， $E \le N + N \times M + M$ 。
- 最坏情况 $N=1000, M=500$ ， $E \approx 5 \times 10^5$ ，在二分下仍可接受。

```cpp
// Jamie's Contact Groups

#include <bits/stdc++.h>
using namespace std;

const int INF = 0x3f3f3f3f;
const int MAXN = 1505; // N <= 1000, M <= 500, 总节点数 N+M+2
int N, M;

struct Edge {
    int to, cap, rev;
};

vector<Edge> graph[MAXN];
int level[MAXN];
int iter[MAXN];

void addEdge(int from, int to, int cap) {
    graph[from].push_back({to, cap, (int)graph[to].size()});
    graph[to].push_back({from, 0, (int)graph[from].size() - 1});
}

// BFS 构建分层图
void bfs(int s) {
    memset(level, -1, sizeof(level));
    queue<int> q;
    level[s] = 0;
    q.push(s);
    while (!q.empty()) {
        int v = q.front(); q.pop();
        for (auto &e : graph[v])
            if (e.cap > 0 && level[e.to] < 0) {
                level[e.to] = level[v] + 1;
                q.push(e.to);
            }
    }
}

// DFS 寻找增广路
int dfs(int v, int t, int f) {
    if (v == t) return f;
    for (int &i = iter[v]; i < (int)graph[v].size(); i++) {
        Edge &e = graph[v][i];
        if (e.cap > 0 && level[v] < level[e.to]) {
            int d = dfs(e.to, t, min(f, e.cap));
            if (d > 0) {
                e.cap -= d;
                graph[e.to][e.rev].cap += d;
                return d;
            }
        }
    }
    return 0;
}

// Dinic 算法求最大流
int maxFlow(int s, int t) {
    int flow = 0;
    while (true) {
        bfs(s);
        if (level[t] < 0) return flow;
        memset(iter, 0, sizeof(iter));
        int f;
        while ((f = dfs(s, t, INF)) > 0) flow += f;
    }
}

// 检查当限制最大组大小为 limit 时是否可行
bool check(int limit, const vector<vector<int>> &can) {
    // 清空图
    for (int i = 0; i < MAXN; i++) graph[i].clear();
    int s = N + M, t = s + 1;
    // 源点 -> 每个联系人，容量 1
    for (int i = 0; i < N; i++) addEdge(s, i, 1);
    // 每个联系人 -> 允许的组，容量 1
    for (int i = 0; i < N; i++)
        for (int g : can[i]) addEdge(i, N + g, 1);
    // 每个组 -> 汇点，容量 limit
    for (int i = 0; i < M; i++) addEdge(N + i, t, limit);
    int flow = maxFlow(s, t);
    return flow == N; // 是否所有人都分配成功
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    while (cin >> N >> M && (N || M)) {
        cin.ignore(); // 忽略第一行末尾的换行符
        vector<vector<int>> can(N); // can[i] 存储第 i 个联系人可以去的组
        for (int i = 0; i < N; i++) {
            string line;
            getline(cin, line);
            stringstream ss(line);
            string name;
            ss >> name; // 名字，本题不需要使用
            int g;
            while (ss >> g) can[i].push_back(g);
        }

        // 二分答案
        int left = 0, right = N, ans = N;
        while (left <= right) {
            int mid = (left + right) / 2;
            if (check(mid, can)) {
                ans = mid;
                right = mid - 1;
            } else left = mid + 1;
        }
        cout << ans << "\n";
    }
    return 0;
}
```

## 总结

本题的关键在于将 **“最小化最大组人数”** 转化为 **二分答案 + 最大流可行性检查** 。通过构建二分图并限制每组容量，可以高效判断某个 $limit$ 是否可行。 $\texttt{Dinic}$ 算法在此类单位容量图上表现良好，可以在时限内解决问题。
