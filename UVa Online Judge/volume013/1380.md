[UVa 1380 A Scheduling Problem](https://blog.csdn.net/metaphysis/article/details/158350726)

## 题目分析

本题是一个任务调度问题，每个任务需要 $1$ 天完成，任务之间存在两种约束：

1. **冲突约束**：某些任务不能在同一天进行（无向边）
2. **先后顺序约束**：某些任务必须在其他任务完成后才能开始（有向边）

题目保证在忽略边的方向后，整个图是一棵树。我们需要求出完成所有任务所需的最少天数。

已知结论：最少天数要么是 $k$，要么是 $k+1$，其中 $k$ 是最长有向路径的顶点数。

## 解题思路

### 核心思想

我们可以将问题转化为**约束满足问题**：

- 每条最长有向路径上的任务已经确定了相对顺序，即路径上第 $i$ 个任务必须在第 $i$ 天完成
- 其他任务需要在 $1$ 到 $k$ 天中选择一个时间
- 有向边 $u \rightarrow v$ 要求 $time[u] < time[v]$
- 无向边 $u - v$ 要求 $time[u] \neq time[v]$

如果能找到一种在 $k$ 天内的安排方案，答案就是 $k$，否则就是 $k+1$。

### 算法步骤

#### 1. 建图

根据输入格式构建三种边：

- `d`：建立有向边 $father \rightarrow son$
- `u`：建立有向边 $son \rightarrow father$
- 无标记：记录无向边 $father - son$

#### 2. 求最长有向路径长度 $k$

使用记忆化搜索计算以每个节点为起点的最长路径长度：

$$
dp[u] = 1 + \max_{v \in out[u]} dp[v]
$$

全局最大值 $k = \max(dp[i])$。

#### 3. 收集所有长度为 $k$ 的最长路径

从所有 $dp[i] = k$ 的节点出发，沿着出边收集路径。

#### 4. 约束传播算法

我们使用**可能颜色集合** $pos[i]$ 表示节点 $i$ 可能的颜色（天数），初始为 $\{1,2,\ldots,k\}$。

**步骤 1**：固定最长路径上的节点颜色
- 对于每条最长路径，第 $i$ 个节点强制颜色为 $i$
- 如果同一节点被强制赋予不同颜色，则冲突

**步骤 2**：约束传播
重复以下过程直到没有变化：
- 检查所有有向边约束
- 检查所有无向边约束
- 如果有节点的可能颜色集合缩小到 $1$ 个，则固定该颜色

**步骤 3**：判断结果
- 如果所有节点都能被赋予颜色，则答案为 $k$
- 否则答案为 $k+1$

### 约束检查详解

#### 有向边 $u \rightarrow v$

- 如果 $u$ 和 $v$ 都已固定：检查 $color[u] < color[v]$
- 如果只有 $u$ 固定：从 $pos[v]$ 中删除所有 $\leq color[u]$ 的颜色
- 如果只有 $v$ 固定：从 $pos[u]$ 中删除所有 $\geq color[v]$ 的颜色
- 如果都未固定：检查是否存在 $c_u \in pos[u]$ 和 $c_v \in pos[v]$ 使得 $c_u < c_v$

#### 无向边 $u - v$

- 如果 $u$ 和 $v$ 都已固定：检查 $color[u] \neq color[v]$
- 如果只有 $u$ 固定：从 $pos[v]$ 中删除 $color[u]$
- 如果只有 $v$ 固定：从 $pos[u]$ 中删除 $color[v]$
- 如果都未固定：检查是否存在 $c_u \in pos[u]$ 和 $c_v \in pos[v]$ 使得 $c_u \neq c_v$

### 复杂度分析

- 最长路径查找：$O(n)$
- 约束传播：最多进行 $O(nk)$ 轮，每轮检查所有边
- 总复杂度：$O(n^2k)$，在 $n \leq 200, k \leq 200$ 时可接受

// A Scheduling Problem

#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> outE, inE;
vector<pair<int,int>> undE;
vector<int> dp, col;
vector<vector<int>> paths;
vector<set<int>> pos;
int n, k;
bool changed;

int dfs(int u) {
    if (dp[u] != -1) return dp[u];
    dp[u] = 1;
    for (int v : outE[u]) dp[u] = max(dp[u], 1 + dfs(v));
    return dp[u];
}

void getPaths(int u, vector<int>& p) {
    p.push_back(u);
    if (p.size() == k) paths.push_back(p);
    else for (int v : outE[u]) if (dp[v] == k - p.size()) getPaths(v, p);
    p.pop_back();
}

bool check() {
    changed = false;
    // 有向边
    for (int u = 1; u <= n; u++) for (int v : outE[u]) {
        if (col[u] != -1 && col[v] != -1) { if (col[u] >= col[v]) return false; continue; }
        if (col[u] != -1) {
            for (int c = 1; c <= col[u]; c++) if (pos[v].erase(c)) changed = true;
            if (pos[v].empty()) return false;
        } else if (col[v] != -1) {
            for (int c = col[v]; c <= k; c++) if (pos[u].erase(c)) changed = true;
            if (pos[u].empty()) return false;
        } else {
            bool ok = false;
            for (int cu : pos[u]) for (int cv : pos[v]) if (cu < cv) { ok = true; break; }
            if (!ok) return false;
        }
    }
    // 无向边
    for (auto& e : undE) {
        int a = e.first, b = e.second;
        if (col[a] != -1 && col[b] != -1) { if (col[a] == col[b]) return false; continue; }
        if (col[a] != -1) {
            if (pos[b].erase(col[a])) changed = true;
            if (pos[b].empty()) return false;
        } else if (col[b] != -1) {
            if (pos[a].erase(col[b])) changed = true;
            if (pos[a].empty()) return false;
        } else {
            bool ok = false;
            for (int ca : pos[a]) for (int cb : pos[b]) if (ca != cb) { ok = true; break; }
            if (!ok) return false;
        }
    }
    return true;
}

bool solve() {
    col.assign(n + 1, -1);
    pos.assign(n + 1, set<int>());
    for (int i = 1; i <= n; i++) for (int c = 1; c <= k; c++) pos[i].insert(c);
    
    for (auto& p : paths) for (int i = 0; i < p.size(); i++) {
        int node = p[i], c = i + 1;
        if (col[node] != -1 && col[node] != c) return false;
        if (col[node] == -1) {
            col[node] = c;
            pos[node] = {c};
        }
    }
    
    do {
        changed = false;
        if (!check()) return false;
        for (int i = 1; i <= n; i++) if (col[i] == -1 && pos[i].size() == 1) {
            col[i] = *pos[i].begin();
            changed = true;
        }
    } while (changed);
    
    return true;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    string line;
    while (getline(cin, line)) {
        if (line == "0") break;
        
        outE.assign(210, vector<int>());
        inE.assign(210, vector<int>());
        undE.clear();
        set<int> vertices;
        
        while (true) {
            if (line == "0") break;
            stringstream ss(line);
            int f; ss >> f;
            vertices.insert(f);
            string token;
            while (ss >> token) {
                if (token == "0") break;
                int s; char t = 0;
                if (token.back() == 'd' || token.back() == 'u') {
                    t = token.back();
                    token.pop_back();
                    s = stoi(token);
                } else s = stoi(token);
                vertices.insert(s);
                if (t == 'd') { outE[f].push_back(s); inE[s].push_back(f); }
                else if (t == 'u') { outE[s].push_back(f); inE[f].push_back(s); }
                else undE.push_back({f, s});
            }
            getline(cin, line);
        }
        
        n = *max_element(vertices.begin(), vertices.end());
        dp.assign(n + 1, -1);
        paths.clear();
        k = 0;
        for (int i = 1; i <= n; i++) k = max(k, dfs(i));
        for (int i = 1; i <= n; i++) if (dp[i] == k) {
            vector<int> p;
            getPaths(i, p);
        }
        cout << (solve() ? k : k + 1) << "\n";
    }
    return 0;
}
